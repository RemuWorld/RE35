cmake_minimum_required(VERSION 3.16)
project(command_environment_bridge VERSION 0.3.2 LANGUAGES C CXX)
find_package(Threads REQUIRED)
# The loader and installed core must use the same source list. Reject drift
# during configuration instead of producing a bridge missing a Lua dependency.
set(CE_CORE_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/native/core_modules.def")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CE_CORE_MANIFEST}")
file(STRINGS "${CE_CORE_MANIFEST}" CE_CORE_LINES)
set(CE_CORE_PATHS)
set(CE_CORE_NAMES)
foreach(CE_CORE_LINE IN LISTS CE_CORE_LINES)
  if(CE_CORE_LINE MATCHES "^CE_CORE_MODULE\\(\"([a-zA-Z0-9_.]+)\", \"([a-zA-Z0-9_/]+\\.lua)\"\\)$")
    set(CE_CORE_NAME "${CMAKE_MATCH_1}")
    set(CE_CORE_PATH "${CMAKE_MATCH_2}")
    if(CE_CORE_NAME IN_LIST CE_CORE_NAMES OR CE_CORE_PATH IN_LIST CE_CORE_PATHS)
      message(FATAL_ERROR "Duplicate core manifest entry: ${CE_CORE_NAME}")
    endif()
    string(REPLACE "/" "." CE_EXPECTED_NAME "${CE_CORE_PATH}")
    string(REGEX REPLACE "\\.lua$" "" CE_EXPECTED_NAME "${CE_EXPECTED_NAME}")
    if(NOT CE_CORE_NAME STREQUAL CE_EXPECTED_NAME OR NOT CE_CORE_PATH MATCHES "^command_environment/core/")
      message(FATAL_ERROR "Invalid core manifest entry: ${CE_CORE_LINE}")
    endif()
    list(APPEND CE_CORE_NAMES "${CE_CORE_NAME}")
    list(APPEND CE_CORE_PATHS "${CE_CORE_PATH}")
  elseif(NOT CE_CORE_LINE MATCHES "^(/\\*.*\\*/|[ \t]*)$")
    message(FATAL_ERROR "Invalid core manifest line: ${CE_CORE_LINE}")
  endif()
endforeach()
file(GLOB_RECURSE CE_ACTUAL_CORE CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/lua"
  "${CMAKE_CURRENT_SOURCE_DIR}/lua/command_environment/core/*.lua")
list(SORT CE_CORE_PATHS)
list(SORT CE_ACTUAL_CORE)
if(NOT CE_CORE_PATHS OR NOT "${CE_CORE_PATHS}" STREQUAL "${CE_ACTUAL_CORE}")
  message(FATAL_ERROR "Core manifest does not match the available Lua sources")
endif()
add_library(command_environment SHARED native/ce.c)
if(WIN32)
  set_target_properties(command_environment PROPERTIES PREFIX "")
endif()
target_include_directories(command_environment PUBLIC native)
target_compile_features(command_environment PRIVATE c_std_11)
target_link_libraries(command_environment PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
add_executable(ce_lua_console native/lua_console.c)
target_compile_features(ce_lua_console PRIVATE c_std_11)
target_link_libraries(ce_lua_console PRIVATE ${CMAKE_DL_LIBS})
add_executable(ce_cpp examples/cpp/main.cpp)
target_compile_features(ce_cpp PRIVATE cxx_std_17)
target_link_libraries(ce_cpp PRIVATE command_environment Threads::Threads)
add_executable(ce_benchmark examples/cpp/benchmark.cpp)
target_compile_features(ce_benchmark PRIVATE cxx_std_17)
target_link_libraries(ce_benchmark PRIVATE command_environment Threads::Threads)
add_executable(ce_stdio native/stdio.c)
target_compile_features(ce_stdio PRIVATE c_std_11)
target_link_libraries(ce_stdio PRIVATE command_environment)
if(MINGW)
  target_link_options(ce_stdio PRIVATE -municode)
endif()
# Host-only process ownership is optional and does not change the core bridge.
if(WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(ce_process_scope native/process_scope.c)
  target_compile_features(ce_process_scope PRIVATE c_std_11)
  if(WIN32)
    target_link_libraries(ce_process_scope PRIVATE shell32)
  endif()
  if(MSVC)
    target_compile_options(ce_process_scope PRIVATE /W4)
  else()
    target_compile_options(ce_process_scope PRIVATE -Wall -Wextra -Werror)
  endif()
  install(TARGETS ce_process_scope)
endif()
if(MSVC)
  target_compile_options(command_environment PRIVATE /W4)
else()
  target_compile_options(command_environment PRIVATE -Wall -Wextra -Werror)
  target_compile_options(ce_lua_console PRIVATE -Wall -Wextra -Werror)
  target_compile_options(ce_cpp PRIVATE -Wall -Wextra -Werror)
  target_compile_options(ce_stdio PRIVATE -Wall -Wextra -Werror)
endif()
install(TARGETS command_environment ce_lua_console ce_cpp ce_stdio)
install(FILES native/ce.h DESTINATION include/command_environment)
foreach(CE_CORE_PATH IN LISTS CE_CORE_PATHS)
  get_filename_component(CE_CORE_DIRECTORY "${CE_CORE_PATH}" DIRECTORY)
  install(FILES "lua/${CE_CORE_PATH}" DESTINATION "share/command_environment/lua/${CE_CORE_DIRECTORY}")
endforeach()
install(FILES LICENSE DESTINATION share/licenses/RE35)
